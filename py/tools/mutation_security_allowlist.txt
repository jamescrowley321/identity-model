# Content-addressed equivalent-mutant allowlist for `make mutation-security`
# (Epic 19 G.1).
#
# mutmut occasionally emits an *equivalent* mutant: a syntactic change that is
# semantically identical to the original and therefore impossible to kill with
# any test. Waive one here, keyed on the CONTENT of the mutation — the function
# it lives in plus the source line it rewrote — NOT its `__mutmut_N` index.
# mutmut numbers mutants sequentially and renumbers them whenever a function
# changes, so an index-keyed waiver could silently rebind to a *different, real*
# survivor and pass the gate green (issue #615: fail-open). Content keys track
# the exact mutation they were written for.
#
# Format — one waiver per line, `#` comment lines (above the waiver) justify it:
#
#   <func_id> | <original line> => <mutated line>   EXACT: only this mutation.
#   <func_id> | <original line>                     LINE:  ANY mutation of this
#                                                          line — use ONLY for a
#                                                          line with no control-
#                                                          flow / security effect
#                                                          (e.g. a log message).
#
# `<func_id>` is the mutmut mutant name up to (not including) `__mutmut_` — e.g.
# `py_identity_model.aio.token_validation.x__refresh_jwks`. Source text matches
# stripped (leading indentation ignored), as `mutmut show` renders it. A line
# with no ` | ` separator is a hard error: the gate must never silently drop a
# waiver it cannot parse.
#
# Every entry weakens the gate, so keep the justification concrete and specific.
# `mutmut show <name>` prints a survivor's exact original/mutated lines.

# ── R.9 issuer pinning (_enforce_allowed_issuers) ─────────────────────────────
# The `else` branch sets `candidates = []`; the mutant sets it to `None`. Both
# fall straight into `if not candidates:` — `not []` and `not None` are both
# True — so both raise the identical "no issuer was resolved"
# InvalidIssuerException and the for-loop below is never reached. No input that
# reaches the else branch can observe [] vs None: equivalent, unkillable.
py_identity_model.core.token_validation_logic.x__enforce_allowed_issuers | candidates = [] => candidates = None

# ── #562 find_key_by_kid: no-kid signing-key kty pre-filter ───────────────────
# The guard is `if len(signing_keys) > 1 and jwt_alg:` and the block only ever
# *narrows* signing_keys to the alg-matching kty. Both mutants change *when* the
# block runs, never its effect:
#   `> 1 or jwt_alg`  — additionally enters when len == 1 (narrowing a single key
#     to itself is a no-op) or when len > 1 & jwt_alg is None (`_ALG_TO_KTY.get(
#     None)` is None, so the inner `if expected_kty:` skips).
#   `>= 1 and jwt_alg` — signing_keys is always >= 1 after the sig fallback, so
#     `>= 1` is always True: identical to `> 1` for every len > 1 case, and the
#     extra len == 1 case narrows a single key to itself (no-op).
# No input can observe a different selected key or algorithm: unkillable.
py_identity_model.core.parsers.x_find_key_by_kid | if len(signing_keys) > 1 and jwt_alg: => if len(signing_keys) > 1 or jwt_alg:
py_identity_model.core.parsers.x_find_key_by_kid | if len(signing_keys) > 1 and jwt_alg: => if len(signing_keys) >= 1 and jwt_alg:

# ── T299 cache-observability: lock-stripe distribution (aio) ──────────────────
# `_get_*_fetch_lock(None)` makes `hash(None) % STRIPES` pick one fixed valid
# stripe instead of hashing the URI/key. Single-flight correctness is unaffected
# — same-URI fetches still serialise on the same lock object — only which of the
# stripes is selected changes, which is not observable.
py_identity_model.aio.token_validation.x__get_disco_response | fetch_lock = _get_disco_fetch_lock(cache_key) => fetch_lock = _get_disco_fetch_lock(None)
py_identity_model.aio.token_validation.x__refresh_jwks | fetch_lock = _get_jwks_fetch_lock(jwks_uri) => fetch_lock = _get_jwks_fetch_lock(None)

# ── Log-only lines (class 1): equivalent under ANY mutation ───────────────────
# These are human-readable log statements — no control flow, return value, or
# counter depends on them, so every mutation mutmut makes to them (message text,
# logger args, uppercasing) is equivalent. Asserting exact log text is brittle
# and out of scope. LINE waivers (no `=>`) waive all mutations of the line.
#
# "Forcing JWKS refresh ..." info line (sync + aio _refresh_jwks):
py_identity_model.sync.token_validation.x__refresh_jwks | logger.info("Forcing JWKS refresh for %s (possible key rotation)", jwks_uri)
py_identity_model.aio.token_validation.x__refresh_jwks | logger.info("Forcing JWKS refresh for %s (possible key rotation)", jwks_uri)
# Empty-keys retained-cache fallback info line (aio _refresh_jwks), one waiver
# per concatenated-string segment mutmut mutates independently:
py_identity_model.aio.token_validation.x__refresh_jwks | "JWKS refresh for %s returned 200 with empty keys; "
py_identity_model.aio.token_validation.x__refresh_jwks | "falling back to retained cache entry for in-flight "
py_identity_model.aio.token_validation.x__refresh_jwks | "validation",

# ── #585 injectable discovery policy: equivalent residuals ────────────────────
# The injectable-policy change threads a `policy` argument through the sync/aio
# cached refresh + retry paths; the behavioural threading is proven by
# test_discovery_policy_rotation_paths.py. The mutants below are the residual
# *equivalent* ones on those changed lines, in two classes.
#
# Class A — default-None no-op. `if policy is None: policy = DiscoveryPolicy()`
# mutated to `policy = None`. `policy` is then used solely as
# `get_jwks(JwksRequest(address=..., policy=policy))`, and get_jwks re-defaults
# `jwks_request.policy or DiscoveryPolicy()` — so `None` and a fresh
# `DiscoveryPolicy()` produce the identical request. Unobservable.
py_identity_model.sync.token_validation.x__get_cached_jwks | policy = DiscoveryPolicy() => policy = None
py_identity_model.sync.token_validation.x__refresh_jwks | policy = DiscoveryPolicy() => policy = None
#
# Class B — require_https arg → None on the retry-path policy resolution:
# `resolve_discovery_policy(config.discovery_policy, config.require_https)` with
# the 2nd arg mutated to None. That arg is only consulted when
# `discovery_policy is None`, and then only as `DiscoveryPolicy(require_https=
# <arg>)`, whose sole retry-path effect is `if not policy.require_https: return`
# in validate_url_scheme — where `not None` and `not False` are both truthy
# (identical admit). The one config that would distinguish them
# (`require_https=True`, discovery_policy=None) cannot reach the retry over an
# http scheme, because the initial discovery/JWKS fetch under that same strict
# config already rejects it. Unobservable.
py_identity_model.sync.token_validation.x__retry_with_refreshed_jwks | token_validation_config.require_https, => None,
py_identity_model.aio.token_validation.x__retry_with_refreshed_jwks | token_validation_config.require_https, => None,
